\section*{A bounded-weight calculation}
\setcounter{lemma}{1}
\begin{lemma}[Bounded reweighting]
Assume $0 \leq w(X) \leq B$ and $0 \leq \ell(X) \leq 1$. Then
\[\mathbb{E}[w(X)\ell(X)] \leq B\mathbb{E}[\ell(X)].\]
\end{lemma}
\begin{proof}
Multiply the pointwise inequality by the nonnegative loss and take expectations.
\end{proof}
